    <script>
        // 授業データの定義
        const courseData = [
            { category: "数学", name: "数と式", count: 6 }, // 1~5 + 修了
            { category: "数学", name: "集合と論証", count: 4 }, // 1~3 + 修了
            { category: "数学", name: "2次関数", count: 6 }, // 1~5 + 修了
            { category: "数学", name: "図形と計量", count: 6 }, // 1~5 + 修了
            { category: "数学", name: "場合の数", count: 5 }, // 1~4 + 修了
            { category: "数学", name: "確率", count: 4 }, // 1~3 + 修了
            { category: "数学", name: "図形の性質", count: 5 }, // 1~4 + 修了
            { category: "数学", name: "データの分析", count: 4 }, // 1~3 + 修了
            { category: "数学", name: "数学と人間の活動", count: 5 }, // 1~4 + 修了
            { category: "英語", name: "レベルアップ英語 (1~10)", count: 10 },
            { category: "英語", name: "中間テスト", count: 1 },
            { category: "英語", name: "レベルアップ英語 (11~20)", count: 10 }
        ];

        // アプリの状態初期化
        let checkedLessons = JSON.parse(localStorage.getItem('toshin_checkedLessons')) || {};
        let reviews = JSON.parse(localStorage.getItem('toshin_reviews')) || [];
        let goals = JSON.parse(localStorage.getItem('toshin_goals')) || { weeklyLessons: 5, weeklyPlay: 300, monthlyLessons: 20 };

        // タブ切り替え関数
        function switchTab(tabId) {
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            document.getElementById('tab-' + tabId).classList.remove('hidden');

            document.querySelectorAll('nav button').forEach(btn => {
                btn.classList.remove('text-indigo-600', 'font-bold');
                btn.classList.add('text-gray-400');
            });
            const activeBtn = document.getElementById('nav-' + tabId);
            activeBtn.classList.remove('text-gray-400');
            activeBtn.classList.add('text-indigo-600', 'font-bold');

            window.scrollTo(0, 0);
            if(tabId === 'calendar') renderHistory();
            if(tabId === 'home') updateHome();
        }

        // 授業リストの描画
        function renderLessons() {
            const container = document.getElementById('lesson-list');
            container.innerHTML = '';
            let totalLessons = 0;
            let completedLessons = 0;

            courseData.forEach((course, cIndex) => {
                let courseCard = document.createElement('div');
                courseCard.className = 'bg-white p-4 rounded-xl shadow-sm border border-gray-100';
                
                let html = `<h3 class="font-bold text-gray-700 mb-2">${course.name} <span class="text-xs font-normal text-gray-400">(${course.count}コマ)</span></h3><div class="grid grid-cols-5 gap-2">`;
                
                for (let i = 1; i <= course.count; i++) {
                    totalLessons++;
                    const key = `${cIndex}_${i}`;
                    const isChecked = checkedLessons[key] ? 'bg-indigo-600 text-white border-indigo-600' : 'bg-gray-50 text-gray-600 border-gray-200';
                    const label = (course.name.includes("修了") || (course.name.includes("中間テスト") && i === 1) || (i === course.count && course.name.includes("修了"))) ? `修了` : `${i}`;
                    
                    // 特殊な名前調整（例：修了コマなどのわかりやすさ）
                    let displayName = i;
                    if (course.count > 1 && i === course.count && course.name.includes("レベルアップ") === false && course.name !== "中間テスト") {
                        displayName = "修了";
                    }

                    if(checkedLessons[key]) completedLessons++;

                    html += `<button onclick="toggleLesson(${cIndex}, ${i})" class="py-2 text-xs font-medium rounded-lg border ${isChecked} transition flex flex-col items-center justify-center">
                        <span>${displayName === "修了" ? "修了" : i}</span>
                    </button>`;
                }
                html += `</div>`;
                courseCard.innerHTML = html;
                container.appendChild(courseCard);
            });

            // 全体進捗更新
            const percent = totalLessons > 0 ? Math.round((completedLessons / totalLessons) * 100) : 0;
            document.getElementById('total-progress-text').innerText = `完了数: ${completedLessons} / ${totalLessons}コマ`;
            document.getElementById('total-progress-percent').innerText = `${percent}%`;
            document.getElementById('total-progress-bar').style.width = `${percent}%`;
            document.getElementById('lesson-count-badge').innerText = `${completedLessons}/${totalLessons} 完了`;
            
            updateHome();
        }

        // 授業のチェック状態切替
        function toggleLesson(cIndex, i) {
            const key = `${cIndex}_${i}`;
            checkedLessons[key] = !checkedLessons[key];
            localStorage.setItem('toshin_checkedLessons', JSON.stringify(checkedLessons));
            renderLessons();
        }

        // ホーム画面の更新
        function updateHome() {
            const goalDiv = document.getElementById('goal-display');
            goalDiv.innerHTML = `
                <div class="flex justify-between py-1 border-b"><span>今週の授業目標:</span><span class="font-bold text-indigo-600">${goals.weeklyLessons} コマ</span></div>
                <div class="flex justify-between py-1 border-b"><span>今週の遊び上限:</span><span class="font-bold text-indigo-600">${goals.weeklyPlay} 分</span></div>
                <div class="flex justify-between py-1"><span>今月の授業目標:</span><span class="font-bold text-indigo-600">${goals.monthlyLessons} コマ</span></div>
            `;
        }

        // 振り返りの保存
        function saveReview(e) {
            e.preventDefault();
            const date = document.getElementById('review-date').value;
            const toshin = parseInt(document.getElementById('toshin-time').value) || 0;
            const other = parseInt(document.getElementById('other-time').value) || 0;
            const play = parseInt(document.getElementById('play-time').value) || 0;

            const newReview = { date, toshin, other, play };
            
            // 同じ日付があれば上書き、なければ追加
            const existingIndex = reviews.findIndex(r => r.date === date);
            if (existingIndex >= 0) {
                reviews[existingIndex] = newReview;
            } else {
                reviews.unshift(newReview);
            }

            localStorage.setItem('toshin_reviews', JSON.stringify(reviews));
            alert('本日の振り返りを保存しました！お疲れ様です！');
            document.getElementById('review-form').reset();
            document.getElementById('review-date').valueAsDate = new Date();
        }

        // 目標の保存
        function saveGoal(e) {
            e.preventDefault();
            goals = {
                weeklyLessons: parseInt(document.getElementById('goal-weekly-lessons').value) || 0,
                weeklyPlay: parseInt(document.getElementById('goal-weekly-play').value) || 0,
                monthlyLessons: parseInt(document.getElementById('goal-monthly-lessons').value) || 0
            };
            localStorage.setItem('toshin_goals', JSON.stringify(goals));
            alert('目標を更新しました！');
            switchTab('home');
        }

        // 履歴・カレンダーの描画
        function renderHistory() {
            const container = document.getElementById('history-list');
            if (reviews.length === 0) {
                container.innerHTML = '<p class="text-sm text-gray-400 text-center py-4">まだ記録がありません。「振り返り」タブから入力してください。</p>';
                return;
            }

            let html = '';
            reviews.forEach(r => {
                const totalStudy = r.toshin + r.other;
                html += `
                    <div class="p-3 bg-gray-50 rounded-xl border border-gray-100 text-sm space-y-1">
                        <div class="flex justify-between font-bold text-indigo-600">
                            <span>📅 ${r.date}</span>
                            <span>勉強計: ${totalStudy}分</span>
                        </div>
                        <div class="grid grid-cols-3 gap-2 text-xs text-gray-600 pt-1">
                            <div>東進: <span class="font-semibold text-gray-800">${r.toshin}分</span></div>
                            <div>その他: <span class="font-semibold text-gray-800">${r.other}分</span></div>
                            <div>遊び: <span class="font-semibold text-gray-800">${r.play}分</span></div>
                        </div>
                    </div>
                `;
            });
            container.innerHTML = html;
        }

        // 初期設定
        document.getElementById('review-date').valueAsDate = new Date();
        document.getElementById('goal-weekly-lessons').value = goals.weeklyLessons;
        document.getElementById('goal-weekly-play').value = goals.weeklyPlay;
        document.getElementById('goal-monthly-lessons').value = goals.monthlyLessons;

        renderLessons();
        updateHome();
    </script>
